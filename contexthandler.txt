// AI SYMPTOM CHECKER 

if(context_id == "ai_checker")
{
	if(answers.containsKey("symptoms") && !answers.containsKey("ai_result"))
	{
		symptoms = answers.get("symptoms").get("text");
		payload = {"symptoms":symptoms};
		ai_raw = invokeurl
		[
			url :base_url + "/ai/symptom-check"
			type :POST
			body:payload.toString()
		];
		// Cleanup
		ai_text = ai_raw.toString().trim();
		ai_text = ai_text.replaceAll("```json","");
		ai_text = ai_text.replaceAll("```","");
		ai_text = ai_text.trim();
		
		start_raw = ai_text.indexOf("{");
		end_raw = ai_text.lastIndexOf("}");
		start_str = start_raw.toString();
		end_str = end_raw.toString();
		if(start_str != "-1" && end_str != "-1")
		{
			start_num = start_str.toNumber();
			end_num = end_str.toNumber();
			json_clean = ai_text.substring(start_num,end_num + 1);
		}
		else
		{
			json_clean = ai_text;
		}
		try 
		{
			ai = json_clean.toMap();
		}
		catch (e)
		{
			ai = Map();
		}
		dept = if(ai.containsKey("department"),ai.get("department"),"General Medicine");
		sev = if(ai.containsKey("severity"),ai.get("severity"),"Not specified");
		cond_text = "";
		if(ai.containsKey("possible_conditions"))
		{
			for each  c in ai.get("possible_conditions")
			{
				cond_text = cond_text + "• " + c + "\n";
			}
		}
		rec = if(ai.containsKey("recommended_action"),ai.get("recommended_action"),"");
		report = "🩺 *AI Symptom Analysis*\n";
		report = report + "--------------------------------\n";
		report = report + "🏥 *Recommended Department:* **" + dept + "**\n";
		report = report + "⚠️ *Severity Level:* **" + sev + "**\n\n";
		report = report + "🧠 *Possible Conditions:*\n" + cond_text + "\n";
		report = report + "📝 *Recommended Action:*\n" + rec + "\n";
		report = report + "--------------------------------\n";
		report = report + "💡 *Note:* This is an AI-generated preliminary assessment.\nIf symptoms persist, please consult a healthcare professional.";
		q = {"name":"ai_result","replies":{report}};
		response.put("questions",{q});
		return response;
	}
	if(answers.containsKey("ai_result"))
	{
		response.put("action","end");
		return response;
	}
}
