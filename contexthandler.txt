response = Map();
response.put("action","context");
response.put("context_id", context_id);

base_url = "https://google-calendar-two.vercel.app";

info "BOOK_APPT :: answers = " + answers;

// booking flow
if (context_id == "book_appt")
{
    // ask name
    if (!answers.containsKey("name"))
    {
        q = {"name":"name","replies":{"👤 May I know your full name?"}};
        response.put("questions",{q});
        return response;
    }

    // ask email
    if (!answers.containsKey("email"))
    {
        q = Map();
        q.put("name","email");
        q.put("replies",{"📧 Please enter your email."});

        v = {"format":"email","error":"❌ Please enter a valid email."};
        q.put("validate", v);

        response.put("questions",{q});
        return response;
    }

    // send OTP
    if (answers.containsKey("email") && !answers.containsKey("email_otp"))
    {
        email = answers.get("email").get("text");

        payload = {"email":email};

        invokeurl
        [
            url : base_url + "/send-otp"
            type : POST
            body : payload.toString()
        ];

        q = Map();
        q.put("name","email_otp");
        q.put("replies",{
            "📩 OTP sent to **" + email + "**.",
            "Please enter the 6-digit code."
        });

        response.put("questions",{q});
        return response;
    }

    // verify OTP
    if (answers.containsKey("email_otp") && !answers.containsKey("otp_verified"))
    {
        email = answers.get("email").get("text");
        otp = answers.get("email_otp").get("text");

        payload = {"email":email, "otp":otp};

        verifyResp = invokeurl
        [
            url : base_url + "/verify-otp"
            type : POST
            body : payload.toString()
        ];

        if (verifyResp.containsKey("valid") && verifyResp.get("valid") == true)
        {
            q = Map();
            q.put("name","otp_verified");
            q.put("replies",{
                "✅ Email verified!",
                "Tap Continue to proceed."
            });

            inp = {"type":"select","options":{"Continue"}};
            q.put("input", inp);

            response.put("questions",{q});
            return response;
        }
        else
        {
            response.put("action","reply");
            response.put("replies",{"❌ Wrong OTP. Try again."});
            return response;
        }
    }

    // ask phone
    if (answers.containsKey("otp_verified") && !answers.containsKey("phone"))
    {
        q = Map();
        q.put("name","phone");
        q.put("replies",{"📱 Please enter your mobile number."});

        v = {"format":"phoneno","error":"❌ Invalid mobile number."};
        q.put("validate", v);

        response.put("questions",{q});
        return response;
    }

    // ask date
    if (answers.containsKey("phone") && !answers.containsKey("date"))
    {
        q = Map();
        q.put("name","date");
        q.put("replies",{"📅 Please choose your appointment date."});
        q.put("input",{"type":"calendar"});

        response.put("questions",{q});
        return response;
    }

    // fetch time slots
    if (answers.containsKey("date") && !answers.containsKey("slot"))
    {
        raw = answers.get("date").get("text");
        only = raw.substring(0, raw.indexOf(" "));
        dt = only.toDate("dd-MMM-yyyy");
        dateFinal = dt.toString("yyyy-MM-dd");

        payload = {"date":dateFinal};
        slotResp = invokeurl
        [
            url : base_url + "/slots"
            type : POST
            body : payload.toString()
        ];

        opts = List();

        // morning slots
        if(slotResp.containsKey("morning"))
        {
            for each s in slotResp.get("morning")
            {
                opts.add("🌅 Morning • " + s.get("time") + " | " + s.get("value"));
            }
        }

        // afternoon slots
        if(slotResp.containsKey("afternoon"))
        {
            for each s in slotResp.get("afternoon")
            {
                opts.add("🌞 Afternoon • " + s.get("time") + " | " + s.get("value"));
            }
        }

        // evening slots
        if(slotResp.containsKey("evening"))
        {
            for each s in slotResp.get("evening")
            {
                opts.add("🌆 Evening • " + s.get("time") + " | " + s.get("value"));
            }
        }

        if (opts.isEmpty())
        {
            response.put("action","reply");
            response.put("replies",{"❌ No slots available for this date."});
            return response;
        }

        q = Map();
        q.put("name","slot");
        q.put("replies",{
            "🗓 Available slots for **" + slotResp.get("readableDate") + "**",
            "Please choose your time:"
        });

        q.put("input",{"type":"select","options":opts});

        response.put("questions",{q});
        return response;
    }

    // create payment link
    if (answers.containsKey("slot") && !answers.containsKey("pay_link_sent"))
    {
        name = answers.get("name").get("text");
        email = answers.get("email").get("text");
        phone = answers.get("phone").get("text");

        slotRaw = answers.get("slot").get("text");
        start = slotRaw.substring(slotRaw.indexOf(" | ") + 3);

        payload = {
            "name":name,
            "email":email,
            "phone":phone,
            "start":start,
            "amount":499
        };

        payResp = invokeurl
        [
            url : base_url + "/create-payment-link"
            type : POST
            body : payload.toString()
        ];

        if (payResp.containsKey("pay_url"))
        {
            link = payResp.get("pay_url");

            q = Map();
            q.put("name","pay_link_sent");
            q.put("replies",{
                "💳 Almost done!",
                "Use the link below to complete your payment:",
                link,
                "Your appointment will be confirmed after payment."
            });

            response.put("questions",{q});
            return response;
        }
        else
        {
            response.put("action","reply");
            response.put("replies",{"❌ Could not generate payment link. Try again."});
            return response;
        }
    }
}

// appointment management flow
if(context_id == "my_appts")
{
    // ask email
    if(!answers.containsKey("lookup"))
    {
        q = Map();
        q.put("name","lookup");
        q.put("replies",{"📧 Please enter your email:"});

        v = {"format":"email","error":"❌ Invalid email."};
        q.put("validate", v);

        response.put("questions",{q});
        return response;
    }

    // fetch user appointments
    lookup = answers.get("lookup").get("text");
    payload = {"email":lookup, "phone":lookup};

    listResp = invokeurl
    [
        url : base_url + "/list"
        type : POST
        body : payload.toString()
    ];

    if(listResp.isEmpty())
    {
        response.put("action","end");
        response.put("replies",{"❌ No appointments found for " + lookup});
        return response;
    }

    // show list to user
    if(!answers.containsKey("selected_appt"))
    {
        opts = List();

        for each appt in listResp
        {
            dt = appt.get("start");
            id = appt.get("id");

            label = dt.substring(0,10) + " at " + dt.substring(11,16) + " | " + id;
            opts.add(label);
        }

        q = Map();
        q.put("name","selected_appt");
        q.put("replies",{"📄 Your appointments:", "Choose one:"});
        q.put("input",{"type":"select","options":opts});

        response.put("questions",{q});
        return response;
    }

    // ask user what they want to do
    if(answers.containsKey("selected_appt") && !answers.containsKey("appt_action"))
    {
        selected = answers.get("selected_appt").get("text");
        appt_id = selected.substring(selected.indexOf(" | ") + 3);

        pre = {
            {"name":"appt_id","value":appt_id},
            {"name":"appt_label","value":selected}
        };

        response.put("precollect", pre);

        q = Map();
        q.put("name","appt_action");
        q.put("replies",{
            "📄 Selected appointment:",
            selected,
            "",
            "Choose an option:"
        });

        q.put("input",{
            "type":"select",
            "options":{"Reschedule","Cancel","Back"}
        });

        response.put("questions",{q});
        return response;
    }

    // cancel appointment
    if(answers.containsKey("appt_action") && answers.get("appt_action").get("text") == "Cancel")
    {
        appt_id = "";
        
        if(answers.containsKey("appt_id"))
            appt_id = answers.get("appt_id").get("text");
        else if(answers.containsKey("selected_appt"))
            appt_id = answers.get("selected_appt").get("text").substring(answers.get("selected_appt").get("text").indexOf(" | ") + 3);

        if(appt_id == "")
        {
            response.put("action","reply");
            response.put("replies",{"⚠️ Couldn't find appointment ID. Try again."});
            return response;
        }

        delReq = {"eventId": appt_id};

        invokeurl
        [
            url : base_url + "/delete"
            type : POST
            body : delReq.toString()
        ];

        response.put("action","end");
        response.put("replies",{
            "🗑️ Appointment cancelled.",
            "You can book again anytime."
        });

        return response;
    }

    // go back
    if(answers.containsKey("appt_action") && answers.get("appt_action").get("text") == "Back")
    {
        response.put("action","reply");
        response.put("replies",{"Returning to menu..."});
        response.put("suggestions",{"Booking Services","Manage My Appointments","Help & Tools"});
        return response;
    }

    // start reschedule → send OTP to lookup email
    if(answers.containsKey("appt_action") && answers.get("appt_action").get("text") == "Reschedule" && !answers.containsKey("r_email_otp"))
    {
        email = answers.get("lookup").get("text");

        req = {"email":email};

        invokeurl
        [
            url : base_url + "/send-otp"
            type : POST
            body : req.toString()
        ];

        q = {"name":"r_email_otp","replies":{"📩 OTP sent to **" + email + "**.","Enter OTP to verify rescheduling:"}};
        response.put("questions",{q});
        return response;
    }

    // verify OTP for reschedule
    if(answers.containsKey("r_email_otp") && !answers.containsKey("r_otp_verified"))
    {
        email = answers.get("lookup").get("text");
        otp   = answers.get("r_email_otp").get("text");

        req = {"email":email,"otp":otp};

        verify = invokeurl
        [
            url : base_url + "/verify-otp"
            type : POST
            body : req.toString()
        ];

        if(verify.get("valid") == true)
        {
            q = {
                "name":"r_otp_verified",
                "replies":{"✅ Email verified!","Tap Continue to pick a new date."},
                "input":{"type":"select","options":{"Continue"}}
            };
            response.put("questions",{q});
            return response;
        }
        else
        {
            response.put("action","reply");
            response.put("replies",{"❌ Wrong OTP. Try again."});
            return response;
        }
    }

    // ask new date
    if(answers.containsKey("r_otp_verified") && !answers.containsKey("new_date"))
    {
        q = {"name":"new_date","replies":{"📅 Choose your new date:"},"input":{"type":"calendar"}};
        response.put("questions",{q});
        return response;
    }

    // ask new time
    if(answers.containsKey("new_date") && !answers.containsKey("new_slot"))
    {
        raw = answers.get("new_date").get("text");
        only = raw.substring(0, raw.indexOf(" "));
        dt = only.toDate("dd-MMM-yyyy");
        fixed = dt.toString("yyyy-MM-dd");

        req = {"date":fixed};

        slotResp = invokeurl
        [
            url : base_url + "/slots"
            type : POST
            body : req.toString()
        ];

        opts = List();

        if(slotResp.containsKey("morning"))
        {
            for each s in slotResp.get("morning")
                opts.add("🌅 " + s.get("time") + " | " + s.get("value"));
        }

        if(slotResp.containsKey("afternoon"))
        {
            for each s in slotResp.get("afternoon")
                opts.add("🌞 " + s.get("time") + " | " + s.get("value"));
        }

        if(slotResp.containsKey("evening"))
        {
            for each s in slotResp.get("evening")
                opts.add("🌆 " + s.get("time") + " | " + s.get("value"));
        }

        q = {"name":"new_slot","replies":{"⏰ Choose a new time slot:"},"input":{"type":"select","options":opts}};
        response.put("questions",{q});
        return response;
    }

    // update appointment
    if(answers.containsKey("new_slot"))
    {
        appt_id = "";

        if(answers.containsKey("appt_id"))
            appt_id = answers.get("appt_id").get("text");
        else if(answers.containsKey("selected_appt"))
            appt_id = answers.get("selected_appt").get("text").substring(answers.get("selected_appt").get("text").indexOf(" | ") + 3);

        if(appt_id == null || appt_id == "")
        {
            response.put("action","reply");
            response.put("replies",{"⚠️ Appointment ID not found."});
            return response;
        }

        raw_slot = answers.get("new_slot").get("text");

        if(!raw_slot.contains(" | "))
        {
            response.put("action","reply");
            response.put("replies",{"⚠️ Slot format incorrect."});
            return response;
        }

        new_start = raw_slot.substring(raw_slot.indexOf(" | ") + 3);

        req = {"eventId": appt_id,"start": new_start};

        upd = invokeurl
        [
            url : base_url + "/update"
            type : POST
            body : req.toString()
        ];

        response.put("action","end");
        response.put("replies",{
            "🎉 Appointment rescheduled!",
            "Check your email for confirmation."
        });

        return response;
    }
}

// AI symptom checker
if (context_id == "ai_checker")
{
    // send symptoms to AI
    if (answers.containsKey("symptoms") && !answers.containsKey("ai_result"))
    {
        symptoms = answers.get("symptoms").get("text");
        payload = {"symptoms": symptoms};

        ai_raw = invokeurl
        [
            url : base_url + "/ai/symptom-check"
            type : POST
            body : payload.toString()
        ];

        ai_text = ai_raw.toString().trim();

        try { ai = ai_text.toMap(); }
        catch (e) { ai = Map(); }

        dept         = if(ai.containsKey("department"), ai.get("department"), "General Medicine");
        expected_doc = if(ai.containsKey("expected_doctor"), ai.get("expected_doctor"), "General Physician");
        sev          = if(ai.containsKey("severity"), ai.get("severity"), "Mild");
        action       = if(ai.containsKey("recommended_action"), ai.get("recommended_action"), "Monitor your symptoms.");
        follow       = if(ai.containsKey("followup_suggestion"), ai.get("followup_suggestion"), "See a doctor if symptoms worsen.");

        // list conditions
        cond_str = "";
        if (ai.containsKey("possible_conditions"))
        {
            for each c in ai.get("possible_conditions")
                cond_str = cond_str + "• " + c + "\n";
        }
        else
        {
            cond_str = "• Common viral illness\n";
        }

        // list tips
        tips_str = "";
        if (ai.containsKey("general_health_tips"))
        {
            for each t in ai.get("general_health_tips")
                tips_str = tips_str + "• " + t + "\n";
        }
        else
        {
            tips_str =
                "• Stay hydrated.\n" +
                "• Take enough rest.\n" +
                "• Monitor symptoms regularly.\n";
        }

        final_text =
            "🩺 *AI Symptom Check*\n\n" +
            "🏥 Department: **" + dept + "**\n" +
            "👨‍⚕️ Doctor: **" + expected_doc + "**\n" +
            "⚠️ Severity: **" + sev + "**\n\n" +
            "Possible Conditions:\n" + cond_str + "\n" +
            "What you can do:\n" + action + "\n\n" +
            "Follow-up:\n" + follow + "\n\n" +
            "Health Tips:\n" + tips_str + "\n" +
            "⚠️ This is AI advice. See a doctor if needed.";

        q = {"name":"ai_result","replies",{final_text}};
        response.put("questions",{q});
        return response;
    }

    // end AI flow
    if (answers.containsKey("ai_result"))
    {
        response.put("action","end");
        return response;
    }
}

return response;
