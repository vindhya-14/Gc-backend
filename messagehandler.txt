response = Map();
msg = "";
msg_l = "";
postback = "";
value = "";

// read incoming text message
if(message != null && message.containsKey("text"))
{
    msg = message.get("text");
    msg_l = msg.toLowerCase();
}
// read postback value (button clicks)
if(message != null && message.containsKey("postback"))
{
    postback = message.get("postback");
}
// read widget value
if(message != null && message.containsKey("value"))
{
    value = message.get("value");
}

info "MSG=" + msg + " | POSTBACK=" + postback + " | VALUE=" + value;

// user clicked "<Service> - Book Now"
if(msg_l.contains(" - book now"))
{
    pos = msg.indexOf(" - ");
    service = msg.substring(0,pos);

    response.put("action","context");
    response.put("context_id","book_appt");

    svc = {"name":"dept","value":service};

    q = {"name":"date",
        "replies":{
            "📅 Booking for *" + service + "*. Please choose your appointment date:"
        },
        "input":{"type":"calendar"}
    };

    response.put("precollect",{svc});
    response.put("questions",{q});
    return response;
}

// handle basic greeting
if(msg_l == "hi" || msg_l == "hello" || msg_l == "hey" || msg_l == "start")
{
    response.put("action","reply");
    response.put("replies",{
        "🌿 Welcome to **CliniQ Assist**.\nHow may I help you today?"
    });

    response.put("suggestions",{
        "Booking Services",
        "Manage My Appointments",
        "Help & Tools"
    });
    return response;
}

// show booking category menu
if(msg_l == "booking services")
{
    response.put("action","reply");
    response.put("replies",{
        "🏥 *Appointment Booking Services*\nChoose an option:"
    });

    response.put("suggestions",{
        "Book Appointment",
        "⬅ Back to Main Menu"
    });

    return response;
}

// show list of departments as cards
if(msg_l == "book appointment")
{
    response.put("action","reply");

    widget = Map();
    widget.put("type","multiple-product");
    widget.put("text","🏥 Please choose a department to continue:");

    items = List();
    placeholder = "https://cdn.pixabay.com/photo/2024/02/16/14/56/man-8577744_1280.jpg";

    depts = {
        {"General Medicine","Fever • Cold • Viral","General Medicine"},
        {"Pediatrics","Child care • Vaccinations","Pediatrics"},
        {"Dermatology","Skin allergies • Acne","Dermatology"},
        {"Orthopedics","Bones • Joints • Injuries","Orthopedics"},
        {"Gynecology","Women's health • Pregnancy","Gynecology"},
        {"ENT","Sinus • Ear pain • Throat","ENT"},
        {"Cardiology","Heart • BP • ECG","Cardiology"},
        {"Neurology","Headache • Seizures","Neurology"},
        {"Ophthalmology","Vision • Eye strain","Ophthalmology"},
        {"Physiotherapy","Pain relief • Rehab","Physiotherapy"}
    };

    for each d in depts
    {
        p = Map();
        p.put("title", d.get(0));
        p.put("subtitle", d.get(1));
        p.put("image", placeholder);

        act = {
            "label":"Book Now",
            "type":"postback",
            "name":"book_service",
            "value":d.get(2)
        };

        p.put("actions",{act});
        items.add(p);
    }

    widget.put("elements",items);
    response.put("replies",{widget});
    response.put("suggestions",{"⬅ Back to Main Menu"});
    return response;
}

// handle user clicking "Book Now" inside the card
if(postback == "book_service")
{
    response.put("action","context");
    response.put("context_id","book_appt");

    pre = {
        {"name":"dept","value":value}
    };

    q = {
        "name":"date",
        "replies":{
            "📅 Booking for *" + value + "*. Please select your preferred date:"
        },
        "input":{"type":"calendar"}
    };

    response.put("precollect",pre);
    response.put("questions",{q});
    return response;
}

// start appointment lookup flow
if(msg_l == "manage my appointments" || msg_l == "my appointments" || msg_l == "manage appointments")
{
    response.put("action","context");
    response.put("context_id","my_appts");

    response.put("questions",{
        {"name":"lookup","replies":{"🔍 Enter your booking email:"}}
    });

    return response;
}

// open help + tools area
if(msg_l == "help & tools")
{
    response.put("action","reply");

    response.put("replies",{
        "🛠 *Help & Tools*\nChoose a tool:"
    });

    response.put("suggestions",{
        "AI Symptom Checker",
        "⬅ Back to Main Menu"
    });

    return response;
}

// start AI symptom checker flow
if(msg_l == "ai symptom checker")
{
    response.put("action","context");
    response.put("context_id","ai_checker");

    q = {
        "name":"symptoms",
        "replies":{
            "🩺 Please describe your symptoms:",
            "Example: *fever, headache, chest pain*"
        }
    };

    response.put("questions",{q});
    return response;
}

// default fallback message when bot can't understand
response.put("action","reply");
response.put("replies",{
    "❓ I didn’t understand. Please choose an option:"
});

response.put("suggestions",{
    "Booking Services",
    "Manage My Appointments",
    "Help & Tools"
});

return response;
